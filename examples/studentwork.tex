\newsavebox\NKUStudentTextBox
\newsavebox\NKUStudentPhotoBox
\newcommand\NKUstudentfit[5]{%
 \begin{pgfinterruptpicture}%
  \sbox\NKUStudentTextBox{\parbox[t]{#2mm}{%
   \setlength\parskip{0pt}\raggedright\noindent #4#5\par}}%
  \ifdim\dimexpr\ht\NKUStudentTextBox+\dp\NKUStudentTextBox\relax>#3mm
   \PackageError{Nankai}{Student work #1 exceeds its text area}{Shorten the field; retain the current font size.}\fi
  \global\setbox\NKUStudentTextBox=\copy\NKUStudentTextBox
 \end{pgfinterruptpicture}}
% Photos fill their frame without distortion.
\newcommand\NKUstudentphoto[5]{%
 \begin{scope}[shift={(#1,#2)}]
  \draw[NKUGray!20,line width=.3pt] (-.45,-.45) rectangle ({#3+.45},{#4+.45});
  \IfFileExists{#5}{%
   \begin{pgfinterruptpicture}%
    \sbox\NKUStudentPhotoBox{\includegraphics[width=#3mm]{#5}}%
    \ifdim\dimexpr\ht\NKUStudentPhotoBox+\dp\NKUStudentPhotoBox\relax<#4mm
     \sbox\NKUStudentPhotoBox{\includegraphics[height=#4mm]{#5}}\fi
    \global\setbox\NKUStudentPhotoBox=\copy\NKUStudentPhotoBox
   \end{pgfinterruptpicture}%
   \begin{scope}
    \clip (0,0) rectangle (#3,#4);
    % Keep faces near the top; remove most overflow from the lower foreground.
    \pgfmathsetmacro\studentphototop{-.15*((\the\ht\NKUStudentPhotoBox+\the\dp\NKUStudentPhotoBox)/1mm-#4)}
    \node[anchor=north,inner sep=0] at ({#3/2},\studentphototop) {\usebox\NKUStudentPhotoBox};
   \end{scope}
  }{\PackageError{Nankai}{Student work photo file not found: #5}{Check the path in NKUStudentGroups.}}%
 \end{scope}}
% Captions belong to their pictures, instead of forming nine loose text blocks.
\newcommand\NKUstudentphotocaption[7]{%
 \NKUstudentphoto{#1}{#2}{#3}{#4}{#5}
 \fill[NKUInk,opacity=.60] (#1,{#2+#4-4.4}) rectangle ({#1+#3},{#2+#4});
 \pgfmathsetmacro\studentcaptionwidth{#3-4}
 \NKUstudentfit{photo caption}{\studentcaptionwidth}{3.9}{\color{white}\bfseries\fontsize{#6}{13}\selectfont}{#7}
 \node[anchor=west,inner sep=0] at ({#1+2},{#2+#4-2.2}) {\usebox\NKUStudentTextBox};}
